21void CodeGenerator::generate_stmt(AST::Stmt& stmt) {
22 if (auto vds = dynamic_cast<AST::VarDeclStmt*>(&stmt)) {
23 generate_var_decl_stmt(*vds);
24 }
25 else if (auto vas = dynamic_cast<AST::VarAsgnStmt*>(&stmt)) {
26 generate_var_asgn_stmt(*vas);
27 }
28 else if (auto fds = dynamic_cast<AST::FuncDeclStmt*>(&stmt)) {
29 generate_func_decl_stmt(*fds);
30 }
31 else if (auto fcs = dynamic_cast<AST::FuncCallStmt*>(&stmt)) {
32 generate_func_call_stmt(*fcs);
33 }
34 else if (auto rs = dynamic_cast<AST::ReturnStmt*>(&stmt)) {
35 generate_return_stmt(*rs);
36 }
37 else if (auto ies = dynamic_cast<AST::IfElseStmt*>(&stmt)) {
38 generate_if_else_stmt(*ies);
39 }
40 else if (auto wcs = dynamic_cast<AST::WhileCycleStmt*>(&stmt)) {
41 generate_while_cycle_stmt(*wcs);
42 }
43 else if (auto dwcs = dynamic_cast<AST::DoWhileCycleStmt*>(&stmt)) {
44 generate_do_while_cycle_stmt(*dwcs);
45 }
46 else if (auto fcs = dynamic_cast<AST::ForCycleStmt*>(&stmt)) {
47 generate_for_cycle_stmt(*fcs);
48 }
49 else if (auto bs = dynamic_cast<AST::BreakStmt*>(&stmt)) {
50 generate_break_stmt(*bs);
51 }
52 else if (auto cs = dynamic_cast<AST::ContinueStmt*>(&stmt)) {
53 generate_continue_stmt(*cs);
54 }
55 else if (auto ms = dynamic_cast<AST::ModuleStmt*>(&stmt)) {
56 generate_module_stmt(*ms);
57 }
58 else if (auto ums = dynamic_cast<AST::UseModuleStmt*>(&stmt)) {
59 generate_use_module_stmt(*ums);
60 }
61 else {
62 throw_exception(SUB_CODEGEN, "Unsupported statement. Please check your Topaz compiler version and fix the problematic section of the code", stmt.line, file_name, is_debug);
63 }

66void CodeGenerator::generate_var_decl_stmt(AST::VarDeclStmt& vds) {
67 llvm::Type *type = type_to_llvm(vds.type);
68 llvm::Value *val = llvm::Constant::getNullValue(type);
69 if (vds.expr != nullptr) {
70 val = generate_expr(*vds.expr);
71 }
72 if (val->getType() != type) {
73 val = implicitly_cast(val, type);
74 }
75 llvm::Value *var = nullptr;
76 if (variables.size() == 1) {
77 var = new llvm::GlobalVariable(*module, type, vds.type.is_const, llvm::GlobalValue::ExternalLinkage, llvm::dyn_cast<llvm::Constant>(val), vds.name);
78 }
79 else {
80 var = builder.CreateAlloca(type, nullptr, vds.name + ".alloca");
81 builder.CreateStore(val, var);
82 }
83 variables.top().emplace(vds.name, var);
84}

86void CodeGenerator::generate_var_asgn_stmt(AST::VarAsgnStmt& vas) {
87 llvm::Value *var_inst = nullptr;
88 auto vars = variables;
89 while (!vars.empty()) {
90 auto vars_it = vars.top().find(vas.name);
91 if (vars_it != vars.top().end()) {
92 var_inst = vars_it->second;
93 }
94 vars.pop();
95 }
96 if (var_inst == nullptr) {
97 std::stringstream ss;
98 ss << "Variable \033[0m'" << vas.name << "'\033[31m does not exists";
99 throw_exception(SUB_CODEGEN, ss.str(), vas.line, file_name, is_debug);
100 }
101 builder.CreateStore(generate_expr(*vas.expr), var_inst);
102}

104void CodeGenerator::generate_func_decl_stmt(AST::FuncDeclStmt& fds) {
105 llvm::Type *ret_type = type_to_llvm(fds.ret_type);
106 std::vector<llvm::Type*> args;
107 std::string func_name = get_mangled_name(fds.name);
108 size_t args_count = fds.args.size();
109 for (size_t i = 0; i < args_count; i++) {
110 args.push_back(type_to_llvm(fds.args[i].type));
111 switch (fds.args[i].type.type) {
112 case AST::TYPE_BOOL:
113 func_name += ".bool";
114 break;
115 case AST::TYPE_CHAR:
116 func_name += ".char";
117 break;
118 case AST::TYPE_SHORT:
119 func_name += ".short";
120 break;
121 case AST::TYPE_INT:
122 func_name += ".int";
123 break;
124 case AST::TYPE_LONG:
125 func_name += ".long";
126 break;
127 case AST::TYPE_FLOAT:
128 func_name += ".float";
129 break;
130 case AST::TYPE_DOUBLE:
131 func_name += ".double";
132 break;
133 case AST::TYPE_TRAIT:
134 func_name += ".T_" + fds.args[i].type.name;
135 break;
136 case AST::TYPE_CLASS:
137 func_name += ".C_" + fds.args[i].type.name;
138 break;
139 }
140 if (fds.args[i].type.is_const) {
141 func_name += "_const";
142 }
143 if (fds.args[i].type.is_ptr) {
144 func_name += "_ptr";
145 }
146 }
147 llvm::FunctionType *func_type = llvm::FunctionType::get(ret_type, args, false);
148 llvm::Function *func = llvm::Function::Create(func_type, llvm::GlobalValue::ExternalLinkage, func_name, *module);
149
150 llvm::BasicBlock *entry = llvm::BasicBlock::Create(context, "entry", func);
151 builder.SetInsertPoint(entry);
152
153 variables.push({});
154 if (functions.find(get_mangled_name(fds.name)) == functions.end()) {
155 functions.emplace(get_mangled_name(fds.name), std::vector<llvm::Function*>{func});
156 }
157 else {
158 functions.at(get_mangled_name(fds.name)).push_back(func);
159 }
160 functions_ret_types.push(ret_type);
161 size_t index = 0;
162 for (llvm::Argument& arg : func->args()) {
163 arg.setName(fds.args[index].name);
164 llvm::AllocaInst *arg_alloca = builder.CreateAlloca(arg.getType(), nullptr, fds.args[index].name);
165 builder.CreateStore(&arg, arg_alloca);
166 variables.top().emplace(fds.args[index].name, arg_alloca);
167 index++;
168 }
169 bool have_ret_in_global = false;
170 for (auto& stmt : fds.block) {
171 if (auto rs = dynamic_cast<AST::ReturnStmt*>(&*stmt)) {
172 have_ret_in_global = true;
173 }
174 generate_stmt(*stmt);
175 }
176 if (!have_ret_in_global) {
177 if (fds.ret_type.type == AST::TYPE_NOTH) {
178 builder.CreateRetVoid();
179 }
180 else if (fds.ret_type.type <= AST::TYPE_DOUBLE) {
181 builder.CreateRet(llvm::Constant::getNullValue(type_to_llvm(fds.ret_type)));
182 }
183 else {
184 throw_exception(SUB_CODEGEN, "Not all paths return a value", fds.line, file_name, is_debug);
185 }
186 }
187 variables.pop();
188 functions_ret_types.pop();
189}

191void CodeGenerator::generate_func_call_stmt(AST::FuncCallStmt& fcs) {
192 std::vector<llvm::Function*> function_candidates = functions.at(get_mangled_name(fcs.name));
193 std::vector<llvm::Value*> args;
194 for (auto& arg : fcs.args) {
195 args.push_back(generate_expr(*arg));
196 }
197
198 size_t last_score = SIZE_MAX;
199 size_t best_candidate_index = 0;
200 size_t candidate_index = 0;
201 for (auto& candidate : function_candidates) {
202 auto candidate_args_it = candidate->args();
203 size_t score = 0;
204 size_t index = 0;
205 for (auto& arg : candidate_args_it) {
206 llvm::Type *candidate_arg_type = arg.getType();
207 llvm::Type *calling_arg_type = generate_expr(*fcs.args[index])->getType();
208
209 if (candidate_arg_type != calling_arg_type) {
210 if (candidate_arg_type->isIntegerTy() && calling_arg_type->isIntegerTy()) {
211 score++;
212 }
213 else if (candidate_arg_type->isFloatingPointTy() && calling_arg_type->isFloatingPointTy()) {
214 score++;
215 }
216 else if (candidate_arg_type->isFloatingPointTy() && calling_arg_type->isIntegerTy()) {
217 score += 2;
218 }
219 else {
220 score += 99;
221 }
222 }
223
224 index++;
225 }
226
227 if (score <= last_score) {
228 best_candidate_index = candidate_index;
229 last_score = score;
230 }
231
232 candidate_index++;
233 }
234
235 builder.CreateCall(function_candidates[best_candidate_index], args, function_candidates[best_candidate_index]->getName() + ".call");
236}

251void CodeGenerator::generate_if_else_stmt(AST::IfElseStmt& ies) {
252 llvm::Function *parent = builder.GetInsertBlock()->getParent();
253 llvm::Value *cond_val = generate_expr(*ies.cond);
254 llvm::BasicBlock *then_bb = llvm::BasicBlock::Create(context, "if.then", parent);
255 llvm::BasicBlock *else_bb = llvm::BasicBlock::Create(context, "if.else", parent);
256 llvm::BasicBlock *merge_bb = llvm::BasicBlock::Create(context, "if.merge", parent);
257
258 builder.CreateCondBr(cond_val, then_bb, else_bb ? else_bb : merge_bb);
259
260 builder.SetInsertPoint(then_bb);
261 variables.push({});
262 for (auto& stmt : ies.then_block) {
263 generate_stmt(*stmt);
264 }
265 variables.pop();
266
267 if (builder.GetInsertBlock()->getTerminator() == nullptr) {
268 builder.CreateBr(merge_bb);
269 }
270
271 builder.SetInsertPoint(else_bb);
272 variables.push({});
273 for (auto& stmt : ies.else_block) {
274 generate_stmt(*stmt);
275 }
276 variables.pop();
277
278 if (builder.GetInsertBlock()->getTerminator() == nullptr) {
279 builder.CreateBr(merge_bb);
280 }
281 builder.SetInsertPoint(merge_bb);

473llvm::Value *CodeGenerator::generate_expr(AST::Expr& expr) {
474 if (auto lit = dynamic_cast<AST::Literal*>(&expr)) {
475 return generate_literal_expr(*lit);
476 }
477 else if (auto be = dynamic_cast<AST::BinaryExpr*>(&expr)) {
478 return generate_binary_expr(*be);
479 }
480 else if (auto ue = dynamic_cast<AST::UnaryExpr*>(&expr)) {
481 return generate_unary_expr(*ue);
482 }
483 else if (auto ve = dynamic_cast<AST::VarExpr*>(&expr)) {
484 return generate_var_expr(*ve);
485 }
486 else if (auto fce = dynamic_cast<AST::FuncCallExpr*>(&expr)) {
487 return generate_func_call_expr(*fce);
488 }
489 else if (auto oce = dynamic_cast<AST::ChainObjects*>(&expr)) {
490 return generate_obj_chain_expr(*oce);
491 }
492 else {
493 throw_exception(SUB_CODEGEN, "An unsupported expression was encountered during compilation. Please check your Topaz compiler version and fix the problematic section of the code", expr.line, file_name, is_debug);
494 }

497llvm::Value *CodeGenerator::generate_literal_expr(AST::Literal& lit) {
498 auto& value = lit.value.value;
499
500 switch (lit.type.type) {
501 case AST::TYPE_CHAR:
502 return llvm::ConstantInt::get(type_to_llvm(lit.type), llvm::APInt(8, std::get<char8_t>(value)));
503 case AST::TYPE_SHORT:
504 return llvm::ConstantInt::get(type_to_llvm(lit.type), llvm::APInt(16, std::get<int16_t>(value)));
505 case AST::TYPE_INT:
506 return llvm::ConstantInt::get(type_to_llvm(lit.type), llvm::APInt(32, std::get<int32_t>(value)));
507 case AST::TYPE_LONG:
508 return llvm::ConstantInt::get(type_to_llvm(lit.type), llvm::APInt(64, std::get<int64_t>(value)));
509 case AST::TYPE_FLOAT:
510 return llvm::ConstantFP::get(type_to_llvm(lit.type), llvm::APFloat(std::get<float_t>(value)));
511 case AST::TYPE_DOUBLE:
512 return llvm::ConstantFP::get(type_to_llvm(lit.type), llvm::APFloat(std::get<double_t>(value)));
513 case AST::TYPE_BOOL:
514 return llvm::ConstantInt::get(type_to_llvm(lit.type), llvm::APInt(1, std::get<bool>(value)));
515 case AST::TYPE_STRING_LIT: {
516 llvm::Constant *str_const = llvm::ConstantDataArray::getString(context, std::get<std::string>(value), true);
517 llvm::GlobalVariable *str_var = new llvm::GlobalVariable(*module, str_const->getType(), true, llvm::GlobalValue::PrivateLinkage, str_const, "string.lit");
518 return str_var;
519 }
520 default:
521 throw_exception(SUB_CODEGEN, "An unsupported literal type was encountered during compilation. Please check your Topaz compiler version and fix the problematic section of the code", lit.line, file_name, is_debug);
522 }
523}

525llvm::Value *CodeGenerator::generate_binary_expr(AST::BinaryExpr& be) {
526 llvm::Value *left = generate_expr(*be.left_expr);
527 llvm::Type *left_type = left->getType();
528 llvm::Value *right = generate_expr(*be.right_expr);
529 llvm::Type *right_type = right->getType();
530
531 llvm::Type *common_type = get_common_type(left_type, right_type);
532 if (left_type != common_type) {
533 left = implicitly_cast(left, common_type);
534 left_type = left->getType();
535 }
536 else if (right_type != common_type) {
537 right = implicitly_cast(right, common_type);
538 right_type = right->getType();
539 }
540 switch (be.op.type) {
541 case TOK_OP_PLUS:
542 if (left_type->isFloatingPointTy() || right_type->isFloatingPointTy()) {
543 return builder.CreateFAdd(left, right, "fadd.tmp");
544 }
545 return builder.CreateAdd(left, right, "add.tmp");
546 case TOK_OP_MINUS:
547 if (left_type->isFloatingPointTy() || right_type->isFloatingPointTy()) {
548 return builder.CreateFSub(left, right, "fsub.tmp");
549 }
550 return builder.CreateSub(left, right, "sub.tmp");
551 case TOK_OP_MULT:
552 if (left_type->isFloatingPointTy() || right_type->isFloatingPointTy()) {
553 return builder.CreateFMul(left, right, "fmul.tmp");
554 }
555 return builder.CreateMul(left, right, "mul.tmp");
556 case TOK_OP_DIV:
557 if (left_type->isFloatingPointTy() || right_type->isFloatingPointTy()) {
558 return builder.CreateFDiv(left, right, "fdiv.tmp");
559 }
560 return builder.CreateSDiv(left, right, "div.tmp");
561 case TOK_OP_MODULO:
562 if (left_type->isFloatingPointTy() || right_type->isFloatingPointTy()) {
563 return builder.CreateFRem(left, right, "frem.tmp");
564 }
565 return builder.CreateSRem(left, right, "rem.tmp");
566 case TOK_OP_EQ_EQ:
567 if (left_type->isFloatingPointTy() || right_type->isFloatingPointTy()) {
568 return builder.CreateFCmpUEQ(left, right, "feq.tmp");
569 }
570 return builder.CreateICmpEQ(left, right, "eq.tmp");
571 case TOK_OP_NOT_EQ_EQ:
572 if (left_type->isFloatingPointTy() || right_type->isFloatingPointTy()) {
573 return builder.CreateNeg(builder.CreateFCmpUEQ(left, right, "feq.tmp"), "fnoteq.tmp");
574 }
575 return builder.CreateNeg(builder.CreateICmpEQ(left, right, "eq.tmp"), "noteq.tmp");
576 case TOK_OP_GT:
577 if (left_type->isFloatingPointTy() || right_type->isFloatingPointTy()) {
578 return builder.CreateFCmpUGT(left, right, "fgt.tmp");
579 }
580 return builder.CreateICmpSGT(left, right, "gt.tmp");
581 case TOK_OP_GT_EQ:
582 if (left_type->isFloatingPointTy() || right_type->isFloatingPointTy()) {
583 return builder.CreateFCmpUGE(left, right, "fge.tmp");
584 }
585 return builder.CreateICmpSGE(left, right, "ge.tmp");
586 case TOK_OP_LS:
587 if (left_type->isFloatingPointTy() || right_type->isFloatingPointTy()) {
588 return builder.CreateFCmpULT(left, right, "flt.tmp");
589 }
590 return builder.CreateICmpSLT(left, right, "lt.tmp");
591 case TOK_OP_LS_EQ:
592 if (left_type->isFloatingPointTy() || right_type->isFloatingPointTy()) {
593 return builder.CreateFCmpULE(left, right, "fle.tmp");
594 }
595 return builder.CreateICmpSLE(left, right, "le.tmp");
596 case TOK_OP_L_AND:
597 return builder.CreateLogicalAnd(left, right, "land.tmp");
598 case TOK_OP_L_OR:
599 return builder.CreateLogicalAnd(left, right, "lor.tmp");
600 default:
601 throw_exception(SUB_CODEGEN, "An unsupported binary operator was encountered during compilation. Please check your Topaz compiler version and fix the problematic section of the code", be.line, file_name, is_debug);
602 }

624llvm::Value *CodeGenerator::generate_var_expr(AST::VarExpr& ve) {
625 auto vars = variables;
626 while (!vars.empty()) {
627 auto vars_it = vars.top().find(ve.name);
628 if (vars_it != vars.top().end()) {
629 llvm::Type *type = nullptr;
630 if (auto global = llvm::dyn_cast<llvm::GlobalVariable>(vars_it->second)) {
631 type = global->getValueType();
632 if (functions_ret_types.empty()) {
633 return global->getInitializer();
634 }
635 }
636 else if (auto local = llvm::dyn_cast<llvm::AllocaInst>(vars_it->second)) {
637 type = local->getAllocatedType();
638 }
639 return builder.CreateLoad(type, vars_it->second, ve.name + ".load");
640 }
641 vars.pop();
642 }
643 std::stringstream ss;
644 ss << "Variable \033[0m'" << ve.name << "'\033[31m does not exists";
645 throw_exception(SUB_CODEGEN, ss.str(), ve.line, file_name, is_debug);

648llvm::Value *CodeGenerator::generate_func_call_expr(AST::FuncCallExpr& fce) {
649 auto function_candidates = functions.at(get_mangled_name(fce.name));
650 std::vector<llvm::Value*> args;
651 for (auto& arg : fce.args) {
652 args.push_back(generate_expr(*arg));
653 }
654
655 size_t last_score = SIZE_MAX;
656 size_t best_candidate_index = 0;
657 size_t candidate_index = 0;
658 for (auto& candidate : function_candidates) {
659 auto candidate_args_it = candidate->args();
660 size_t score = 0;
661 size_t index = 0;
662 for (auto& arg : candidate_args_it) {
663 llvm::Type *candidate_arg_type = arg.getType();
664 llvm::Type *calling_arg_type = generate_expr(*fce.args[index])->getType();
665
666 if (candidate_arg_type != calling_arg_type) {
667 if (candidate_arg_type->isIntegerTy() && calling_arg_type->isIntegerTy()) {
668 score++;
669 }
670 else if (candidate_arg_type->isFloatingPointTy() && calling_arg_type->isFloatingPointTy()) {
671 score++;
672 }
673 else if (candidate_arg_type->isFloatingPointTy() && calling_arg_type->isIntegerTy()) {
674 score += 2;
675 }
676 else {
677 score += 99;
678 }
679 }
680
681 index++;
682 }
683
684 if (score <= last_score) {
685 best_candidate_index = candidate_index;
686 last_score = score;
687 }
688
689 candidate_index++;
690 }
691
692 return builder.CreateCall(function_candidates[best_candidate_index], args, function_candidates[best_candidate_index]->getName() + ".call");
693}

695llvm::Value *CodeGenerator::generate_obj_chain_expr(AST::ChainObjects& co) {
696 llvm::Value *value = nullptr; // Value of target object
697 size_t current_path_size = current_path.size();
698 for (size_t i = 0; i < co.chain.size(); i++) {
699 if (auto ve = dynamic_cast<AST::VarExpr*>(&*co.chain[i])) {
700 current_path.push({ve->name, PathPart::OBJ_MODULE});
701 }
702 else if (auto fce = dynamic_cast<AST::FuncCallExpr*>(&*co.chain[i])) {
703 value = generate_func_call_expr(*fce);
704 }
705 }
706 for (size_t i = current_path.size() - current_path_size; i > 0; i--) {
707 current_path.pop();
708 }
709 return value;

735llvm::Type *CodeGenerator::get_common_type(llvm::Type *left, llvm::Type *right) {
736 if (left == right) {
737 return left;
738 }
739 else if (left->isIntegerTy() || right->isIntegerTy()) {
740 if (left->isIntegerTy() && right->isIntegerTy()) {
741 unsigned left_width = left->getIntegerBitWidth();
742 unsigned right_width = right->getIntegerBitWidth();
743
744 return left_width > right_width ? left : right;
745 }
746 else if (left->isFloatingPointTy() || right->isFloatingPointTy()) {
747 return left->isFloatingPointTy() ? left : right;
748 }
749 }
750 else if (left->isFloatingPointTy() && right->isFloatingPointTy()) {
751 return left->isDoubleTy() && right->isFloatTy() ? left : right;
752 }
753 else if (left->isDoubleTy() || right->isDoubleTy()) {
754 return llvm::Type::getDoubleTy(context);
755 }
756 return nullptr;
757}

759llvm::Value *CodeGenerator::implicitly_cast(llvm::Value *val, llvm::Type *expected_type) {
760 llvm::Type *val_type = val->getType();
761 if (val_type == expected_type) {
762 return val;
763 }
764 else if (val_type->isIntegerTy() && expected_type->isIntegerTy()) {
765 unsigned long value_width = val_type->getIntegerBitWidth();
766 unsigned long expected_width = expected_type->getIntegerBitWidth();
767
768 if (value_width > expected_width) {
769 return builder.CreateTrunc(val, expected_type, "trunc.tmp");
770 }
771 else {
772 return builder.CreateSExt(val, expected_type, "sext.tmp");
773 }
774 }
775 else if (val_type->isFloatingPointTy() && expected_type->isFloatingPointTy()) {
776 if (val_type->isFloatTy() && expected_type->isDoubleTy()) {
777 return builder.CreateFPExt(val, expected_type, "fpext.tmp");
778 }
779 else {
780 return builder.CreateFPTrunc(val, expected_type, "fptrunc.tmp");
781 }
782 }
783 else if (val_type->isIntegerTy() && expected_type->isFloatingPointTy()) {
784 return builder.CreateSIToFP(val, expected_type, "sitofp.tmp");
785 }
786 return nullptr;
787}
