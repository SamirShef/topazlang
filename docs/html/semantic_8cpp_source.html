21void SemanticAnalyzer::analyze_stmt(AST::Stmt& stmt) {
22 if (auto vds = dynamic_cast<AST::VarDeclStmt*>(&stmt)) {
23 analyze_var_decl_stmt(*vds);
24 }
25 else if (auto vas = dynamic_cast<AST::VarAsgnStmt*>(&stmt)) {
26 if (current_space != SPACE_FUNCTION) {
27 throw_exception(SUB_SEMANTIC, "Assignment of variable cannot be in global or module space", stmt.line, file_name);
28 }
29 analyze_var_asgn_stmt(*vas);
30 }
31 else if (auto fds = dynamic_cast<AST::FuncDeclStmt*>(&stmt)) {
32 analyze_func_decl_stmt(*fds);
33 }
34 else if (auto fcs = dynamic_cast<AST::FuncCallStmt*>(&stmt)) {
35 if (current_space != SPACE_FUNCTION) {
36 throw_exception(SUB_SEMANTIC, "Calling of function cannot be in global or module space", stmt.line, file_name);
37 }
38 analyze_func_call_stmt(*fcs);
39 }
40 else if (auto rs = dynamic_cast<AST::ReturnStmt*>(&stmt)) {
41 analyze_return_stmt(*rs);
42 }
43 else if (auto ies = dynamic_cast<AST::IfElseStmt*>(&stmt)) {
44 if (current_space != SPACE_FUNCTION) {
45 throw_exception(SUB_SEMANTIC, "Control flow cannot be in global or module space", stmt.line, file_name);
46 }
47 analyze_if_else_stmt(*ies);
48 }
49 else if (auto wcs = dynamic_cast<AST::WhileCycleStmt*>(&stmt)) {
50 if (current_space != SPACE_FUNCTION) {
51 throw_exception(SUB_SEMANTIC, "While loop cannot be in global or module space", stmt.line, file_name);
52 }
53 analyze_while_cycle_stmt(*wcs);
54 }
55 else if (auto dwcs = dynamic_cast<AST::DoWhileCycleStmt*>(&stmt)) {
56 if (current_space != SPACE_FUNCTION) {
57 throw_exception(SUB_SEMANTIC, "Do-while loop cannot be in global or module space", stmt.line, file_name);
58 }
59 analyze_do_while_cycle_stmt(*dwcs);
60 }
61 else if (auto fcs = dynamic_cast<AST::ForCycleStmt*>(&stmt)) {
62 if (current_space != SPACE_FUNCTION) {
63 throw_exception(SUB_SEMANTIC, "For loop cannot be in global or module space", stmt.line, file_name);
64 }
65 analyze_for_cycle_stmt(*fcs);
66 }
67 else if (auto bs = dynamic_cast<AST::BreakStmt*>(&stmt)) {
68 analyze_break_stmt(*bs);
69 }
70 else if (auto cs = dynamic_cast<AST::ContinueStmt*>(&stmt)) {
71 analyze_continue_stmt(*cs);
72 }
73 else if (auto ms = dynamic_cast<AST::ModuleStmt*>(&stmt)) {
74 analyze_module_stmt(*ms);
75 }
76 else if (auto ums = dynamic_cast<AST::UseModuleStmt*>(&stmt)) {
77 analyze_use_module_stmt(*ums);
78 }
79 else {
80 throw_exception(SUB_SEMANTIC, "Unsupported statement. Please check your Topaz compiler version and fix the problematic section of the code", stmt.line, file_name);
81 }

141 Space previous_space = current_space;

143

177 if (val != nullptr) {
178 have_ret_in_global = true;
179 }
180 }
181 else if (auto fcs = dynamic_cast<AST::ForCycleStmt*>(&*stmt)) {
182 Value *val = get_function_return_value_from_for_cycle(*fcs);
183 if (val != nullptr) {

185 }
186 }
187 }
188 if (!have_ret_in_global && ret_type.type != AST::TYPE_NOTH) {
189 std::stringstream ss;

191 throw_exception(SUB_SEMANTIC, ss.str(), fds.line, file_name);
192 }
193 functions_ret_types.pop();
194 variables.pop();

468 if (auto lit = dynamic_cast<AST::Literal*>(&expr)) {

470 }
471 else if (auto be = dynamic_cast<AST::BinaryExpr*>(&expr)) {

484 return analyze_obj_chain_expr(*oce);

498

501

505 right_type.type >= AST::TYPE_BOOL && right_type.type <= AST::TYPE_DOUBLE && left_type.type > AST::TYPE_DOUBLE) {
506 std::stringstream ss;
507 ss << "Type mismatch: it is not possible to use the binary \033[0m'" << be.op.value <<"'\033[31m operator with \033[0m'" << left_type.to_str() << "'\033[31m and \033[0m'" << right_type.to_str() <<"'\033[31m types";
508 throw_exception(SUB_SEMANTIC, ss.str(), be.line, file_name);
509 }
510 else {
511 if (left_type.type == AST::TYPE_STRING_LIT && right_type.type == AST::TYPE_STRING_LIT) {
512 if (be.op.type != TOK_OP_PLUS) {
513 std::stringstream ss;
514 ss << "Type mismatch: it is not possible to use the binary \033[0m'" << be.op.value <<"'\033[31m operator with \033[0m'" << left_type.to_str() << "'\033[31m and \033[0m'" << right_type.to_str() <<"'\033[31m types";
515 throw_exception(SUB_SEMANTIC, ss.str(), be.line, file_name);
516 }
517 return Value(AST::Type(AST::TYPE_STRING_LIT, "string"), std::get<7>(left_val.value.value) + std::get<7>(right_val.value.value), left_val.is_value_unknown || right_val.is_value_unknown, left_val.is_literal && right_val.is_literal);
518 }
519 switch (be.op.type) {
520 #define VALUE(op, type) Value(output_type, static_cast<type>(binary_two_variants(left_val, right_val, op, be.line)), left_val.is_value_unknown || right_val.is_value_unknown, left_val.is_literal && right_val.is_literal)
521 case TOK_OP_PLUS:
522 case TOK_OP_MINUS:
523 case TOK_OP_MULT:
524 case TOK_OP_DIV:
525 case TOK_OP_MODULO:
526 if (be.op.type >= TOK_OP_PLUS && be.op.type <= TOK_OP_MODULO &&
527 (be.op.type != TOK_OP_PLUS || left_type.type != AST::TYPE_STRING_LIT || right_type.type != AST::TYPE_STRING_LIT) &&
528 (left_type.type > AST::TYPE_DOUBLE || right_type.type > AST::TYPE_DOUBLE)) {

531 throw_exception(SUB_SEMANTIC, ss.str(), be.line, file_name);
532 }
533 case TOK_OP_EQ_EQ:
534 case TOK_OP_NOT_EQ_EQ:
535 case TOK_OP_GT:
536 case TOK_OP_GT_EQ:
537 case TOK_OP_LS:
538 case TOK_OP_LS_EQ:
539 if (be.op.type >= TOK_OP_EQ_EQ) {
540 output_type = AST::Type(AST::TYPE_BOOL, "bool");
541 }
542 if (be.op.type > TOK_OP_NOT_EQ_EQ && (left_type.type > AST::TYPE_DOUBLE || left_type.type == AST::TYPE_BOOL || right_type.type > AST::TYPE_DOUBLE || right_type.type == AST::TYPE_BOOL)) {
543 std::stringstream ss;
544 ss << "Type mismatch: it is not possible to use the binary \033[0m'" << be.op.value <<"'\033[31m operator with \033[0m'" << left_type.to_str() << "'\033[31m and \033[0m'" << right_type.to_str() <<"'\033[31m types";
545 throw_exception(SUB_SEMANTIC, ss.str(), be.line, file_name);
546 }
547 case TOK_OP_L_AND:
548 case TOK_OP_L_OR:
549 if (be.op.type >= TOK_OP_L_AND && be.op.type <= TOK_OP_L_OR && (left_type.type != AST::TYPE_BOOL || right_type.type != AST::TYPE_BOOL)) {
550 std::stringstream ss;

552 throw_exception(SUB_SEMANTIC, ss.str(), be.line, file_name);
553 }
554 if (left_val.is_value_unknown || right_val.is_value_unknown) {
555 return Value(output_type, 0, left_val.is_value_unknown || right_val.is_value_unknown, left_val.is_literal && right_val.is_literal);
556 }
557 switch (output_type.type) {
558 case AST::TYPE_BOOL:
559 return VALUE(be.op.type, bool);
560 case AST::TYPE_CHAR:
561 return VALUE(be.op.type, char);
562 case AST::TYPE_SHORT:

564 case AST::TYPE_INT:
565 return VALUE(be.op.type, int);
566 case AST::TYPE_LONG:

570 case AST::TYPE_DOUBLE:
571 return VALUE(be.op.type, double);

573 default: {}
574 #undef VALUE
575 }

725SemanticAnalyzer::Value SemanticAnalyzer::get_function_return_value(FunctionInfo *func, AST::FuncCallExpr& fce) {
726 variables.push({});
727 functions_ret_types.push(func->ret_type);
728 for (size_t i = 0; i < fce.args.size(); i++) {
729 Value val = analyze_expr(*fce.args[i]);
730 val.type = func->args[i].type;
731 variables.top().emplace(func->args[i].name, val);
732 }
733 for (auto& stmt : func->block) {
734 analyze_stmt(*stmt);
735 if (auto rs = dynamic_cast<AST::ReturnStmt*>(&*stmt)) {
736 Value val = analyze_expr(*rs->expr);
737 val = implicitly_cast(val, functions_ret_types.top(), rs->line);
738 variables.pop();
739 functions_ret_types.pop();
740 return val;
741 }
742 else if (auto ies = dynamic_cast<AST::IfElseStmt*>(&*stmt)) {
743 Value *val = get_function_return_value_from_if_else(*ies);
744 if (val != nullptr) {
745 variables.pop();
746 functions_ret_types.pop();
747 return *val;
748 }
749 }
750 else if (auto wcs = dynamic_cast<AST::WhileCycleStmt*>(&*stmt)) {
751 Value *val = get_function_return_value_from_while_cycle(*wcs);
752 if (val != nullptr) {
753 variables.pop();
754 functions_ret_types.pop();
755 return *val;
756 }
757 }
758 else if (auto dwcs = dynamic_cast<AST::DoWhileCycleStmt*>(&*stmt)) {
759 Value *val = get_function_return_value_from_do_while_cycle(*dwcs);
760 if (val != nullptr) {
761 variables.pop();
762 functions_ret_types.pop();
763 return *val;
764 }
765 }
766 else if (auto fcs = dynamic_cast<AST::ForCycleStmt*>(&*stmt)) {
767 Value *val = get_function_return_value_from_for_cycle(*fcs);
768 if (val != nullptr) {
769 variables.pop();
770 functions_ret_types.pop();
771 return *val;
772 }
773 }
774 }
775 std::stringstream ss;
776 ss << "Not all paths returns value in function \033[0m'" << fce.name << "'\033[31m. Please add \033[0m'return'\033[31m statement into the end of the function";
777 throw_exception(SUB_SEMANTIC, ss.str(), fce.line, file_name);
778}

787 val = implicitly_cast(val, functions_ret_types.top(), rs->line);

790 else if (auto ies = dynamic_cast<AST::IfElseStmt*>(&*stmt)) {
791 return get_function_return_value_from_if_else(*ies);

802 }
803 }
804 else if (cond_val.is_literal && !cond_val.is_value_unknown && std::get<bool>(cond_val.value.value) == false) {
805 for (auto& stmt : ies.else_block) {
806 analyze_stmt(*stmt);

983SemanticAnalyzer::Value SemanticAnalyzer::implicitly_cast(Value val, AST::Type type, uint32_t line) {
984 AST::Type output_type = get_common_type(val.type, type, line);
985 Value res = Value(output_type, 0, val.is_value_unknown, val.is_literal);
986
987 double val_from_variant = 0;
988 switch (val.value.value.index()) {
989 case 0:
990 val_from_variant = std::get<0>(val.value.value);
991 break;
992 case 1:
993 val_from_variant = std::get<1>(val.value.value);
994 break;
995 case 2:
996 val_from_variant = std::get<2>(val.value.value);
